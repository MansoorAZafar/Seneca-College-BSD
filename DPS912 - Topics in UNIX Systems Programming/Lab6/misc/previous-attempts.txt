template <class T>
Socket<T> initializeSocket(int domain, int type, int protocol = 0, const SocketRole& role) {
    const int socket_file_descriptor { socket(domain, type, protocol) };
    if(socket_file_descriptor < 0) {
        throw std::runtime_error("\033[31m\033[ERROR][0m: failed to open socket file descriptor\n");
    }

    return { socket_file_descriptor, role };
}

    const struct sockaddr* address = reinterpret_cast<sockaddr*>(socket.address);
    if(bind(socket.socket_file_descriptor, address, sizeof(Address))) {
        throw std::runtime_error("\033[31m[ERROR]\033[31m: failed to bind socket port\n");
    }

    if(listen(socket.socket_file_descriptor, backlog)) {
        throw std::runtime_error("\033[31m[ERROR]\033[31m: failed to listen on socket\n")
    }
}

    struct Request {
        int fd;
        void* buffer;
        size_t size;
        int flags;
        std::optional<sockaddr*> address;
        std::optional<socklen_t*> len;

    };
    
    class SocketCommunicationStrategy {
    public:
        virtual ssize_t write(const Request& req) = 0;
        virtual ssize_t read(const Request& req) = 0;
    };

    class StreamCommunicationStrategy : public SocketCommunicationStrategy {
    public:
        ssize_t write(const Request& req) override {
            return ::send(req.fd, req.buffer, req.size, req.flags);
        }

        ssize_t read(const Request& req) override {
            return ::recv(req.fd, req.buffer, req.size, req.flags);            
        }
    };

    class DgramCommunicationStrategy : public SocketCommunicationStrategy {
        ssize_t write(const Request& req) override {
            return ::sendto(req.fd, req.buffer, req.size, req.flags, *req.address, *(*req.len));
        }

        ssize_t read(const Request& req) override {
            return ::recvfrom(req.fd, req.buffer, req.size, req.flags, *req.address, *req.len);            
        }
    };

    class Socket {
    protected:
        int fd;
        int type;
        std::unique_ptr<SocketCommunicationStrategy> communication;
        
        Socket(const int domain, const int _type, const int prototype)
            : fd(::socket(domain, _type, prototype)), type(_type) {
            if(this->fd == -1) {
                throw std::runtime_error("socket()");
            }

            this->setStrategy(_type);
        }

        explicit Socket(const int _fd, const int _type) : fd(_fd), type(_type) {
            this->setStrategy(_type);
        }
    public:

        void setStrategy(const int _type) {
            if(_type == SOCK_DGRAM) {
                this->communication = std::make_unique<DgramCommunicationStrategy>();
            } else {
                this->communication = std::make_unique<StreamCommunicationStrategy>();
            }
        }

        ssize_t read(const Request& request) {
            return this->communication->read(request);
        }

        ssize_t write(const Request& request) {
            return this->communication->write(request);
        }
        
        // Remove copy operations
        // - we remove it since if original is closed, 
        //   the copy is then referencing a dead fd
        Socket(const Socket&) = delete;
        Socket& operator=(const Socket&) = delete;

        Socket(Socket&& other) {
            *this = std::move(other);
        }

        Socket& operator=(Socket&& other) {
            if(this != &other) {
                if(this->fd == -1) ::close(this->fd);
                this->fd = other.fd;
                other.fd = -1;
            }

            return *this;
        }

        // All sockets should be able to read and write
        virtual ~Socket() {
            if(this->fd != -1) {
                ::close(this->fd);
            }
        }
    };

            *this->peer = _peer;
            *this->len = sizeof(Address);
        }

        ssize_t write(const std::string_view& message, int flags = 0) {
            const sockaddr* address { reinterpret_cast<const sockaddr*>(&(*this->peer)) };

            return ::sendto(
                this->fd, 
                message.data(), 
                message.size(), 
                flags, 
                address,
                *this->len
            );
        }

        ssize_t read(void* buf, const size_t size, int flags = 0) {
            // sockaddr* address = reinterpret_cast<sockaddr*>(&(*this->peer));
            // std::cout << "address: " << (address != nullptr ? address->sa_data : "BOO") << "\n";

            return ::recvfrom(
                this->fd, 
                buf, 
                size, 
                flags, reinterpret_cast<sockaddr*>(&(*this->peer)), 
                &(*this->len)
            );
        }

        void bind() {
            std::cout << "Unlinking: " << this->socketAddress.sockPath << "\n"; 
            ::unlink(this->socketAddress.sockPath.data());

            std::cout << "Binding: " << this->socketAddress.sockPath << "\n";
            if(::bind(
                this->fd,
                reinterpret_cast<const sockaddr*>(&this->socketAddress.address),
                sizeof(Address)
            ) == -1) {
                throw std::runtime_error("bind()");
            }
        }

        ~DatagramSocket() {
            std::cout << "Unlinking: " << this->socketAddress.sockPath << "\n"; 
            ::unlink(this->socketAddress.sockPath.data());
        }
    };

    template <class Address>
    class DatagramServer {
    private:
        DatagramSocket<Address> socket;
    public:
        explicit DatagramServer(const int domain, const SocketAddress<Address>& addr, const int protocol = 0)
            : socket(domain, addr, protocol) {}
        
        DatagramSocket<Address> connect(const Address& _peer = {}) {
            std::cout << "Connecting to peer\n";
            Address peer = _peer;
            return DatagramSocket<Address>(this->socket.fd, peer);
        }

        void bind() {
            this->socket.bind();
        }
    };

   template <class Address>
    class StreamServer {
    private:
        StreamSocket socket;
        SocketAddress<Address> sockAddr;

    public:
        explicit StreamServer(const int domain, const SocketAddress<Address>& addr, const int protocol = 0) 
            : socket(domain, protocol), sockAddr(addr) {}
        
        void bind() {
            if(::bind(this->socket.fd, 
                reinterpret_cast<const sockaddr*>(&this->sockAddr.address), 
                sizeof(Address)
            ) == -1) {
                throw std::runtime_error("bind()");
            }
        }

        void listen(const int backlog = 5) {
            if(::listen(this->socket.fd, backlog) == -1) {
                throw std::runtime_error("listen()");
            }
        }

        std::optional<StreamSocket> accept() {
            socklen_t len { sizeof(Address) };
            sockaddr* addr { reinterpret_cast<sockaddr*>(&this->sockAddr.address) };

            const int newfd { ::accept(this->socket.fd, addr, &len) };
            if(newfd == -1) {
                return std::nullopt;
                // throw std::runtime_error("accept()");
            }

            return std::optional(StreamSocket(newfd));
        }

        void start() {
            std::cout << "Unlinking " << this->sockAddr.sockPath << "\n";
            ::unlink(this->sockAddr.sockPath.data());

            std::cout << "Binding Server on: " << this->sockAddr.sockPath << "\n";
            this->bind();

            std::cout << "Server Listening\n";
            this->listen();
        }

        ~StreamServer() {
            std::cout << "Unlinking " << this->sockAddr.sockPath << "\n";
            ::unlink(this->sockAddr.sockPath.data());
        }
    };

    template <class Address>
    class StreamClient : public StreamSocket {
    private:
        SocketAddress<Address> sockAddress;
    public:
        explicit StreamClient(const int domain, const SocketAddress<Address>& addr, const int protocol = 0)
            : StreamSocket(domain, protocol), sockAddress(addr) {}
        
        void connect() {
            StreamSocket::connect(
                reinterpret_cast<const sockaddr*>(&this->sockAddress.address), 
                sizeof(Address)
            );
        }
    };

namespace MISC {
    void UnixSocket::initializeAddress(const char* socketPath) {
        if(this->socket->role == SocketRole::SERVER) {
            ::unlink(socketPath);
        }

        memset(&this->socket->address, 0, sizeof(struct sockaddr_un));
        this->socket->address.sun_family = AF_UNIX;
        strncpy(this->socket->address.sun_path, socketPath, sizeof(this->socket->address.sun_path) - 1);
    }

    UnixSocket::UnixSocket(const SocketRole &role, const int type, const char* socketPath, const int protocol)
        : socket(role == SocketRole::CLIENT ? 
            static_cast<Socket<sockaddr_un>*>(new Client<sockaddr_un>(AF_UNIX, type, protocol, role)) : 
            static_cast<Socket<sockaddr_un>*>(new Server<sockaddr_un>(AF_UNIX, type, protocol, role))) {
        this->initializeAddress(socketPath);
    }

    int UnixSocket::connect() const {
        const int connection = reinterpret_cast<Client<sockaddr_un>*>(this->socket)->connect();
        return connection;
    }

    ssize_t UnixSocket::write(const std::string_view &message) const {
        
        if(this->socket->role == SocketRole::SERVER) {
            return reinterpret_cast<Server<sockaddr_un>*>(this->socket)->write(message);
        } 

        std::cout << "[Client Socket Write]\n";
        return this->socket->write(message);
    }

    ssize_t UnixSocket::read(char *rawBuffer, const int size) const {
        if(this->socket->role == SocketRole::SERVER) {
            return reinterpret_cast<Server<sockaddr_un>*>(this->socket)->read(rawBuffer, size);
        } 

        return this->socket->read(rawBuffer, size);
    }

    bool UnixSocket::accept() {
        return reinterpret_cast<Server<sockaddr_un>*>(this->socket)->accept();
    }

    void UnixSocket::start(const int &backlog) const {
        reinterpret_cast<Server<sockaddr_un>*>(this->socket)->start(backlog);
    }

    UnixSocket::~UnixSocket() {
        ::close(this->socket->file_descriptor);
        if(this->socket->role == SocketRole::SERVER) {
            ::close(reinterpret_cast<Server<sockaddr_un>*>(this->socket)->client_file_descriptor);
            ::unlink(this->socket->address.sun_path);
        }

        if(this->socket != nullptr) delete this->socket;
    }

namespace MISC {
    SocketAddress<sockaddr_un> unixSocket(const std::string_view& socketPath) {
        sockaddr_un address{};

        address.sun_family = AF_UNIX;
        strncpy(address.sun_path, socketPath.data(), sizeof(address.sun_path) - 1);
        address.sun_path[sizeof(address.sun_path) - 1] = '\0';

        return {address, socketPath};
    }
